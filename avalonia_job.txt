  # ===========================================================================
  # Avalonia — external data collector (no coverlet in repo); MTP test runner
  # ===========================================================================
  Avalonia:
    name: "Coverage: Avalonia"
    runs-on: ubuntu-24.04
    container:
      image: mcr.microsoft.com/dotnet/sdk:10.0-noble
    timeout-minutes: 360
    if: github.event.inputs.repo == 'all' || github.event.inputs.repo == 'Avalonia'
    steps:
      - name: Checkout orchestrator repo (for local composite actions)
        uses: actions/checkout@692973e3d937129bcbf40652eb9f2f61becf3332 # v4.1.7

      - uses: ./.github/actions/setup-workspace

      - name: Clone Avalonia at pinned SHA
        run: |
          git clone --filter=blob:none https://github.com/AvaloniaUI/Avalonia.git target
          cd target
          git fetch --depth 1 origin dcb607179b707a6c0e41a39fc697af43c00abc48
          git checkout FETCH_HEAD
          git submodule update --init --recursive --depth 1

      - uses: ./.github/actions/restore-nuget-cache
        with:
          repo-name: Avalonia
          sha-prefix: dcb60717

      - uses: ./.github/actions/install-dotnet-coverage
        with:
          version: ${{ env.DOTNET_COVERAGE_VERSION }}

      - name: Install coverlet.console
        # Avalonia uses Microsoft.Testing.Platform (global.json sets
        # test.runner=Microsoft.Testing.Platform). Under MTP, the legacy
        # `--collect "Code Coverage;Format=cobertura"` data collector is
        # silently ignored — tests run cleanly but cobertura output is a
        # 169-byte empty stub. Use coverlet.console (mono.cecil-based managed
        # instrumentation, runner-independent) instead. Same pattern as the
        # aspnetcore job. Run 25495265941 / Avalonia: empty cobertura.
        shell: bash
        run: |
          dotnet tool install -g coverlet.console --version 6.0.2

      # Per-csproj restore+build instead of `dotnet restore Avalonia.slnx`.
      # The slnx pulls in src/Android/Avalonia.Android (needs `android` workload)
      # and samples/ControlCatalog.{Browser,iOS} (need `wasm-tools`/iOS workload),
      # which fail with NETSDK1147 on the noble SDK container. Restoring only the
      # *.UnitTests.csproj projects we actually run avoids the workload-bound
      # graph entirely. Run 25527102157 / Avalonia: failed at slnx restore.
      - name: Restore + build UnitTests projects
        shell: bash
        working-directory: target
        run: |
          mapfile -t PROJECTS < <(find tests -type f -name "*.UnitTests.csproj" \
            ! -path "*RenderTests*" \
            ! -path "*LeakTests*" \
            ! -path "*Designer*" \
            ! -path "*Browser*" \
            ! -path "*FunctionalTests*" \
            ! -path "*IntegrationTests*" \
            | sort -u)
          echo "Discovered ${#PROJECTS[@]} test projects:"
          printf '  %s\n' "${PROJECTS[@]}"
          for proj in "${PROJECTS[@]}"; do
            echo "::group::Restore+Build $(basename "$proj")"
            dotnet restore "$proj" \
              || echo "::warning::Restore of $proj failed — continuing"
            dotnet build "$proj" -c Debug --no-restore \
              || echo "::warning::Build of $proj failed — continuing"
            echo "::endgroup::"
          done

      # MTP-aware test invocation via coverlet.console. coverlet wraps dotnet
      # test, instruments the test assembly DLL via mono.cecil, and emits a
      # real cobertura report independent of which test runner runs inside.
      # Find the built test DLL for each csproj under bin/Debug/<tfm>/, then
      # invoke `coverlet "$asm" --target dotnet --targetargs "test ..."`.
      - name: Run tests with coverage (per-csproj, UnitTests subset)
        shell: bash
        working-directory: target
        continue-on-error: true
        run: |
          export PATH="$PATH:/github/home/.dotnet/tools"
          mkdir -p TestResults
          FILTER='FullyQualifiedName!~FunctionalTests&FullyQualifiedName!~IntegrationTests&FullyQualifiedName!~E2E&FullyQualifiedName!~EndToEnd&FullyQualifiedName!~Stress&Category!=Integration&Category!=IntegrationTest&Category!=IntegrationTests&Category!=E2E&Category!=EndToEnd&Category!=Functional&Category!=FunctionalTests&Category!=Stress&Category!=Performance&Category!=Quarantined&Category!=Flaky'
          mapfile -t PROJECTS < <(find tests -type f -name "*.UnitTests.csproj" \
            ! -path "*RenderTests*" \
            ! -path "*LeakTests*" \
            ! -path "*Designer*" \
            ! -path "*Browser*" \
            ! -path "*FunctionalTests*" \
            ! -path "*IntegrationTests*" \
            | sort -u)
          echo "Discovered ${#PROJECTS[@]} test projects"
          i=0
          for proj in "${PROJECTS[@]}"; do
            i=$((i+1))
            base=$(basename "$proj" .csproj)
            proj_dir=$(dirname "$proj")
            out="$(pwd)/TestResults/${i}_${base}.cobertura.xml"
            echo "::group::[$i/${#PROJECTS[@]}] $base"
            # R2 (2026-05-08): pass the bin/Debug/<tfm> DIRECTORY to coverlet,
            # not a single test DLL. With a single-DLL target, coverlet only
            # instruments that one assembly, and under MTP the test runner
            # process loads production DLLs from the same folder uninstrumented
            # — yielding a cobertura with 0 hits across all classes (R1: empty
            # 169-byte stub). Directory mode instruments every DLL in the bin
            # folder, so production assemblies get tracked when the MTP test
            # exe loads them.
            tfm_dir=$(set +o pipefail; find "$proj_dir/bin/Debug" -maxdepth 1 -mindepth 1 -type d 2>/dev/null | head -1 || true)
            if [[ -z "$tfm_dir" || ! -d "$tfm_dir" ]]; then
              echo "::warning::No bin/Debug/<tfm> directory for $base — skipping"
              echo "::endgroup::"
              continue
            fi
            coverlet "$tfm_dir" \
              --target "dotnet" \
              --targetargs "test \"$proj\" --no-build --filter \"$FILTER\"" \
              --format cobertura \
              --output "$out" \
              --include-test-assembly \
              --exclude "[xunit*]*" \
              --exclude "[NUnit*]*" \
              --exclude "[Microsoft.Testing.*]*" \
              --exclude "[Microsoft.VisualStudio.*]*" \
              --exclude "[Moq*]*" \
              || echo "::warning::Tests in $base failed or runner pressure — continuing"
            echo "::endgroup::"
          done
          dotnet-coverage merge -r -f cobertura \
            -o TestResults/coverage.cobertura.xml \
            "TestResults/*.cobertura.xml" || echo "merge produced no output"

      - uses: ./.github/actions/validate-cobertura
        if: always()
        with:
          working-directory: target

      - name: Disk usage (end)
        if: always()
        run: df -h

      - uses: ./.github/actions/cobertura-report-and-upload
        if: always()
        with:
          repo-name: Avalonia
          working-directory: target
          reportgenerator-version: ${{ env.REPORTGENERATOR_VERSION }}
