  # ===========================================================================
  # .NET Runtime — external dotnet-coverage + native build deps
  # ===========================================================================
  runtime:
    name: "Coverage: .NET Runtime"
    runs-on: ubuntu-24.04
    container:
      image: mcr.microsoft.com/dotnet/sdk:10.0-noble
    timeout-minutes: 360
    if: github.event.inputs.repo == 'all' || github.event.inputs.repo == 'runtime'
    steps:
      - name: Checkout orchestrator repo (for local composite actions)
        uses: actions/checkout@692973e3d937129bcbf40652eb9f2f61becf3332 # v4.1.7

      - name: Mark workspace as a safe git directory
        # Inlined (not via setup-workspace) so apt-get can run before the
        # disk-usage probe — installing native deps grows /var measurably.
        run: git config --global --add safe.directory '*'

      - name: Install native build dependencies
        run: |
          apt-get update -qq
          apt-get install -y --no-install-recommends \
            build-essential cmake clang llvm lld lldb \
            libicu-dev liblttng-ust-dev libssl-dev libkrb5-dev libunwind8-dev \
            pigz cpio python3
          rm -rf /var/lib/apt/lists/*

      - name: Disk usage (start)
        run: df -h

      - name: Clone .NET Runtime at pinned SHA
        run: |
          git clone --filter=blob:none https://github.com/dotnet/runtime.git target
          cd target
          git fetch --depth 1 origin 9ffface2f3fa6fbbb427793c3230b1626a1fdd84
          git checkout FETCH_HEAD

      - uses: ./.github/actions/restore-nuget-cache
        with:
          repo-name: runtime
          sha-prefix: 9ffface2

      - uses: ./.github/actions/install-dotnet-coverage
        with:
          version: ${{ env.DOTNET_COVERAGE_VERSION }}

      - name: Build & test runtime libraries (with coverage)
        working-directory: target
        continue-on-error: true
        shell: bash
        run: |
          export PATH="$PATH:/github/home/.dotnet/tools"
          chmod +x build.sh
          # dotnet-coverage follows child processes spawned by build.sh's test runners.
          # libs+libs.tests subset = managed-only build (no native CoreCLR compilation).
          dotnet-coverage collect \
            "./build.sh -subset libs+libs.tests -test" \
            -f cobertura -o ./TestResults/coverage.cobertura.xml

      - name: Disk usage (end)
        if: always()
        run: df -h

      # Runtime keeps custom report+upload steps because its cobertura paths
      # are multi-rooted (TestResults/ + artifacts/**) and the artifact also
      # ships TRX files. The shared composite assumes a single
      # working-directory + glob pair, which doesn't fit.
      - name: Generate HTML report (best-effort)
        if: always()
        shell: bash
        run: |
          dotnet tool install -g dotnet-reportgenerator-globaltool --version ${{ env.REPORTGENERATOR_VERSION }}
          export PATH="$PATH:/github/home/.dotnet/tools"
          shopt -s globstar nullglob
          # Look for coverage in both the dotnet-coverage output AND the artifacts/ tree
          mapfile -t reports < <(find target -type f -name 'coverage.cobertura.xml' 2>/dev/null)
          if [ "${#reports[@]}" -gt 0 ]; then
            reportgenerator \
              -reports:"target/**/coverage.cobertura.xml" \
              -targetdir:"CoverageReport" \
              -reporttypes:Html
          else
            echo "WARN: no cobertura XML produced"
            mkdir -p CoverageReport
          fi

      - name: Upload HTML coverage report
        if: always()
        uses: actions/upload-artifact@b4b15b8c7c6ac21ea08fcf65892d2ee8f75cf882 # v4.4.3
        with:
          name: coverage-report-runtime
          path: CoverageReport/
          retention-days: 30
          if-no-files-found: warn

      - name: Upload raw cobertura XML
        if: always()
        uses: actions/upload-artifact@b4b15b8c7c6ac21ea08fcf65892d2ee8f75cf882 # v4.4.3
        with:
          name: coverage-xml-runtime
          path: |
            target/TestResults/coverage.cobertura.xml
            target/artifacts/**/coverage.cobertura.xml
            target/artifacts/TestResults/**/*.trx
          retention-days: 90
          if-no-files-found: warn
