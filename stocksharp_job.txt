  # Switch to coverlet.console (mono.cecil instrumentation, runner-agnostic).
  # Same pattern as Avalonia / eShop. Built assembly is StockSharp.Tests.dll
  # under Tests/bin/Release/net10.0/. The Tests project pulls
  # Microsoft.Data.SqlClient + Ecng.Data.SqlServer; SQL-bound test cases will
  # fail in isolation and are filtered by Category=Integration / Stress etc.
  # No private NuGet feed config (default nuget.org for Ecng.* packages).
  StockSharp:
    name: "Coverage: StockSharp"
    runs-on: ubuntu-24.04
    container:
      image: mcr.microsoft.com/dotnet/sdk:10.0-noble
    timeout-minutes: 360
    if: github.event.inputs.repo == 'all' || github.event.inputs.repo == 'StockSharp'
    steps:
      - name: Checkout orchestrator repo (for local composite actions)
        uses: actions/checkout@692973e3d937129bcbf40652eb9f2f61becf3332 # v4.1.7

      - uses: ./.github/actions/setup-workspace

      - name: Clone StockSharp at pinned SHA
        run: |
          git clone --filter=blob:none https://github.com/StockSharp/StockSharp.git target
          cd target
          git fetch --depth 1 origin a26ce597cab73fe07949e0cc3845216342d33484
          git checkout FETCH_HEAD

      - uses: ./.github/actions/restore-nuget-cache
        with:
          repo-name: StockSharp
          sha-prefix: a26ce597

      - uses: ./.github/actions/install-dotnet-coverage
        with:
          version: ${{ env.DOTNET_COVERAGE_VERSION }}

      - name: Restore + build Tests project
        shell: bash
        working-directory: target
        # Per-csproj restore + build (StockSharp_Tests.slnx pulls 22 projects
        # including BasketSecurity etc.; restoring the whole graph adds drag.
        # Tests.csproj has the ProjectReferences it needs and will pull them
        # transitively).
        run: |
          dotnet restore Tests/Tests.csproj
          dotnet build Tests/Tests.csproj -c Release --no-restore

      - name: Run tests with coverage (in-process data collector)
        shell: bash
        working-directory: target
        continue-on-error: true
        # R5 (2026-05-08): switch to in-process `dotnet test --collect "Code
        # Coverage;Format=cobertura"`, matching the proven abp/efcore/roslyn
        # pattern. Tests.csproj uses Microsoft.NET.Test.Sdk + MSTest.TestAdapter
        # (classic vstest, not MTP — confirmed in common_target_tests.props),
        # so the built-in Microsoft.CodeCoverage data collector loads inside
        # testhost at startup and avoids the wrap-mode CLR-profiler-attach
        # failure that produced 178-byte stubs in R1 / R4 (runs 25552132370,
        # 25577060633) and the empty-Module bug in R2 / R3 with coverlet.
        #
        # Flaky-test exclusions retained from R2: AsyncExtensionsTests /
        # ConnectorBasketTests fire off-by-one event-count assertions on hosted
        # Linux runners; PathsTests asserts on host paths; ReportTests compares
        # byte-exact .xml/.csv; TransactionIdStorageTests InMemory case is
        # timing-sensitive.
        run: |
          export PATH="$PATH:/github/home/.dotnet/tools"
          mkdir -p TestResults
          dotnet test Tests/Tests.csproj --no-build -c Release \
            --filter "FullyQualifiedName!~FunctionalTests&FullyQualifiedName!~IntegrationTests&FullyQualifiedName!~E2E&FullyQualifiedName!~EndToEnd&FullyQualifiedName!~Stress&FullyQualifiedName!~AsyncExtensionsTests&FullyQualifiedName!~ConnectorBasketTests&FullyQualifiedName!~PathsTests&FullyQualifiedName!~ReportTests&FullyQualifiedName!~TransactionIdStorageTests&Category!=Integration&Category!=IntegrationTest&Category!=IntegrationTests&Category!=E2E&Category!=EndToEnd&Category!=Functional&Category!=FunctionalTests&Category!=Stress&Category!=Performance&Category!=Quarantined&Category!=Flaky" \
            --collect "Code Coverage;Format=cobertura" \
            --results-directory TestResults \
            || echo "::warning::Tests in StockSharp.Tests failed or runner pressure — continuing"
          dotnet-coverage merge -r -f cobertura \
            -o TestResults/coverage.cobertura.xml \
            "TestResults/*.cobertura.xml" "TestResults/**/*.cobertura.xml" \
            || echo "merge produced no output"

      - uses: ./.github/actions/validate-cobertura
        if: always()
        with:
          working-directory: target

      - name: Disk usage (end)
        if: always()
        run: df -h

      - uses: ./.github/actions/cobertura-report-and-upload
        if: always()
        with:
          repo-name: StockSharp
          working-directory: target
          reportgenerator-version: ${{ env.REPORTGENERATOR_VERSION }}

    needs:
